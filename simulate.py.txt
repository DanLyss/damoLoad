#!/usr/bin/env python3
"""
Module 3: Symmetrical 10-Channel Parametric Waveform Simulator.
Version 12.0: Clean C-Ready Physics Architecture.
Eliminated SciPy dependencies. Replaced ARMA(1,1)+SkewNormal with a cascade of 
two AR(1) processes and centered Split-Normal innovations. Fixed systematic mean drift, 
exact cascade variance normalization, and energy-conserving weight distribution.
"""

import argparse
import json
import os
import sys
import numpy as np

def parse_arguments():
    parser = argparse.ArgumentParser(description="Module 3 Simulator: Generates synthetic 10-channel trajectories.")
    parser.add_argument("--passport", type=str, required=True, help="Path to 10-channel JSON passport")
    parser.add_argument("--meta", type=str, required=True, help="Path to system metadata passport")
    parser.add_argument("--output-traj", type=str, required=True, help="Path for output synthetic CSV trajectory")
    parser.add_argument("--seed", type=int, default=0, help="Random seed for deterministic runs")
    parser.add_argument("-v", "--verbose", action="store_true", help="Enable verbose tracing")
    parser.add_argument("--steps", type=int, default=50, help="Number of time steps to simulate")
    parser.add_argument("--legacy-clip", action="store_true", help="Enable legacy hard upper-bound clipping for M_raw (flat ceiling)")
    parser.add_argument("--time-step-ratio", type=float, default=1.0, help="Ratio of simulation time step to passport time step (dt)")
    return parser.parse_args()

def main():
    args = parse_arguments()
    np.random.seed(args.seed)
    dt = args.time_step_ratio

    if not os.path.exists(args.meta) or not os.path.exists(args.passport):
        print(f"[CRITICAL] Required configuration missing.", file=sys.stderr)
        sys.exit(1)

    with open(args.meta, 'r', encoding='utf-8') as f:
        meta = json.load(f)
    with open(args.passport, 'r', encoding='utf-8') as f:
        passport = json.load(f)

    t_sim = args.steps
    m_cols = int(meta['matrix_geometry']['cols'])
    
    # Изменено на 10 каналов для EM-3
    P_hat = np.zeros((t_sim, 10))
    
    channel_keys = [
        "w1", "mu1", "sigma1", 
        "w2", "mu2", "sigma2", 
        "w3", "mu3", "sigma3", 
        "M_raw"
    ]
    
    # Пересчитаны индексы для 3 компонент
    weight_indices = [0, 3, 6]
    center_indices = [1, 4, 7]
    width_indices = [2, 5, 8]

    if args.verbose:
        print(f"[*] Starting 10-channel synthesis: {t_sim} steps (dt={dt}).")

    # Инициализация векторов состояния фильтров под 10 каналов
    v_states = np.zeros(10)
    z_states = np.zeros(10)

    # Предрасчет временных меток
    t_indices = np.arange(t_sim, dtype=float) * dt

    for p, v_key in enumerate(channel_keys):
        if v_key not in passport:
            print(f"[ERROR] Channel key {v_key} not found in passport configuration.", file=sys.stderr)
            sys.exit(1)
            
        m = passport[v_key]
        
        k, b = m['trend_slope'] * dt, m['trend_intercept']
        sigma_x, gamma_x = m['std_dev'], m['skewness']
        f_dom, a_dom, phi_start = m['f_dom'], m['a_dom'], m['phi_start']
        
        outlier_rate = m.get('outlier_rate', 0.0) * dt
        
        # --- Инициализация параметров физического каскада фильтров ---
        rho_0 = m.get('autocorrelation_lag1', 0.8)
        lam = -np.log(max(1e-4, min(rho_0, 0.9999)))
        
        # Коэффициенты затухания для двух AR(1) звеньев
        phi_v = np.exp(-lam * dt)
        phi_z = np.exp(-lam * dt)
        
        # Линейный перевод skewness в параметр асимметрии Split-Normal шума
        alpha = np.clip(gamma_x * 0.5, -0.9, 0.9)
        
        # Расчет точных теоретических моментов Split-Normal для компенсации дрейфа и дисперсии
        mu_eps = alpha * np.sqrt(2.0 / np.pi)
        s2 = 1.0 + (alpha**2) * (1.0 - 2.0 / np.pi)
        
        # Вычисление точной теоретической дисперсии каскада двух AR(1) звеньев
        denom1 = max(1e-12, 1.0 - phi_v**2)
        denom2 = max(1e-12, 1.0 - phi_z**2)
        denom3 = max(1e-12, 1.0 - phi_v * phi_z)
        var_z = (dt**2 * s2 * (1.0 + phi_v * phi_z)) / (denom1 * denom2 * denom3)
        std_z = np.sqrt(var_z)

        # Аналитический масштабный коэффициент для поправки под временной шаг dt
        num_val = dt * lam - 1.0 + np.exp(-dt * lam)
        den_val = lam - 1.0 + np.exp(-lam)
        scale_factor = np.sqrt((num_val / (dt**2)) / max(1e-12, den_val))

        # Теоретическая дисперсия суммарной флуктуации (гармоника + нормированный шум)
        var_harmonic_theoretical = (a_dom ** 2) / 2.0
        theo_fluct_var = var_harmonic_theoretical + (sigma_x * scale_factor)**2
        theo_fluct_std = np.sqrt(theo_fluct_var) if theo_fluct_var > 1e-6 else 1e-6

        # --- Последовательный цикл генерации по шагам времени ---
        for t_idx in range(t_sim):
            t_curr = t_indices[t_idx]
            
            # 1. Сплит-нормальная инновация с мгновенным центрированием (убираем дрейф baseline)
            xi = np.random.normal(0, 1)
            epsilon_raw = xi * (1.0 + alpha) if xi >= 0.0 else xi * (1.0 - alpha)
            epsilon = epsilon_raw - mu_eps
            
            # 2. Обновление каскада СДУ (вместо нестабильной ARMA)
            v_states[p] = phi_v * v_states[p] + epsilon
            z_states[p] = phi_z * z_states[p] + dt * v_states[p]
            
            # Стандартизация случайной компоненты к строгому единичному отклонению
            z_scaled = z_states[p] / std_z
            
            # 3. Расчет детерминированных компонент (оптимизировано, без делений в цикле)
            trend = b + k * t_idx
            harmonic = a_dom * np.cos(2.0 * np.pi * f_dom * t_curr + phi_start)
            
            # 4. Линейная сборка и балансировка суммарной энергии флуктуаций
            raw_fluct = harmonic + (sigma_x * scale_factor) * z_scaled
            fluct_clean = (raw_fluct / theo_fluct_std) * sigma_x
            X_raw = trend + fluct_clean
            
            # 5. Генерация аномалий (outliers) в рантайме с защитой от смещений
            if outlier_rate > 0.0 and np.random.binomial(1, np.clip(outlier_rate, 0.0, 1.0)):
                outlier_mean = m.get('outlier_mean', 0.0)
                outlier_std = m.get('outlier_std', 0.0)
                outlier_skewness = m.get('outlier_skewness', 0.0)
                
                xi_out = np.random.normal(0, 1)
                alpha_out = np.clip(outlier_skewness * 0.5, -0.9, 0.9)
                eps_out_raw = xi_out * (1.0 + alpha_out) if xi_out >= 0.0 else xi_out * (1.0 - alpha_out)
                
                # Центрирование и нормализация шума выброса
                mu_out = alpha_out * np.sqrt(2.0 / np.pi)
                var_out = 1.0 + (alpha_out**2) * (1.0 - 2.0 / np.pi)
                eps_out_clean = (eps_out_raw - mu_out) / np.sqrt(var_out)
                
                outlier_val = outlier_mean + outlier_std * eps_out_clean
                threshold_bound = 3.0 * (sigma_x * scale_factor)
                
                if outlier_mean > 0:
                    outlier_val = max(outlier_val, threshold_bound)
                elif outlier_mean < 0:
                    outlier_val = min(outlier_val, -threshold_bound)
                    
                X_raw += outlier_val

            # 6. Пространственный гео-клиппинг
            if p in weight_indices: 
                P_hat[t_idx, p] = np.clip(X_raw, 0.0, 1.0)
            elif p in center_indices: 
                P_hat[t_idx, p] = np.clip(X_raw, 0.0, float(m_cols))
            elif p in width_indices: 
                P_hat[t_idx, p] = np.clip(X_raw, 0.5, float(m_cols) / 2.0)
            else:
                max_m_raw = float(m_cols * 9.0) if args.legacy_clip else None
                P_hat[t_idx, p] = np.clip(X_raw, 0.0, max_m_raw)

    # --- Пост-обработка: Энергосберегающая L1-нормировка долей весов ---
    current_weight_sum = np.sum(P_hat[:, weight_indices], axis=1)
    current_weight_sum = np.where(current_weight_sum < 1e-12, 1e-12, current_weight_sum)
    for idx in weight_indices:
        P_hat[:, idx] /= current_weight_sum

    # Экспорт результатов в CSV (адаптировано под 10 каналов)
    output_matrix = np.hstack((t_indices.reshape(-1, 1), P_hat))
    np.savetxt(
        args.output_traj, 
        output_matrix, 
        fmt=['%.6f'] + ['%.6f'] * 10, 
        delimiter=',',
        header="time," + ",".join(channel_keys), 
        comments=''
    )
    
    meta['time_step_ratio'] = dt
    with open(args.meta, 'w', encoding='utf-8') as f:
        json.dump(meta, f, indent=4)

    if args.verbose:
        print(f"[SUCCESS] Module 3 simulation saved to '{args.output_traj}'.")

if __name__ == "__main__":
    main()
